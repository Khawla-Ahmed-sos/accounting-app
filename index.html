import sqlite3 from datetime import datetime class PharmacyDeliveryERP: def __init__(self, db_path=":memory:"): # استخدام :memory: يعمل بسلاسة في بيئات التشغيل السحابية والهاتف # غيّره إلى "pharmacy_erp.db" إذا كنت تريد حفظ البيانات على الملفات self.conn = sqlite3.connect(db_path) self.cursor = self.conn.cursor() self._create_tables() self._seed_chart_of_accounts() def _create_tables(self): self.cursor.executescript(""" CREATE TABLE IF NOT EXISTS accounts ( code TEXT PRIMARY KEY, name TEXT NOT NULL, acc_type TEXT NOT NULL CHECK(acc_type IN ('ASSET', 'LIABILITY', 'EQUITY', 'REVENUE', 'EXPENSE')), normal_balance TEXT NOT NULL CHECK(normal_balance IN ('DEBIT', 'CREDIT')) ); CREATE TABLE IF NOT EXISTS journal_entries ( id INTEGER PRIMARY KEY AUTOINCREMENT, entry_date TEXT NOT NULL, reference TEXT NOT NULL, description TEXT NOT NULL ); CREATE TABLE IF NOT EXISTS journal_lines ( id INTEGER PRIMARY KEY AUTOINCREMENT, entry_id INTEGER NOT NULL, account_code TEXT NOT NULL, debit REAL NOT NULL DEFAULT 0.0, credit REAL NOT NULL DEFAULT 0.0, FOREIGN KEY (entry_id) REFERENCES journal_entries(id), FOREIGN KEY (account_code) REFERENCES accounts(code) ); """) self.conn.commit() def _seed_chart_of_accounts(self): # 1. دليل الحسابات (Chart of Accounts) coa = [ # الأصول (مدين بطبيعته) ("1101", "الصندوق والبنوك / المحافظ", "ASSET", "DEBIT"), ("1102", "ذمم العملاء المدينة", "ASSET", "DEBIT"), ("1103", "ذمم الصيدليات المدينة", "ASSET", "DEBIT"), # الخصوم (دائن بطبيعته) ("2101", "مستحقات الصيدليات (دائنون)", "LIABILITY", "CREDIT"), ("2102", "مستحقات مناديب التوصيل", "LIABILITY", "CREDIT"), # حقوق الملكية (دائن بطبيعته) ("3101", "رأس المال", "EQUITY", "CREDIT"), # الإيرادات (دائن بطبيعته) ("4101", "إيراد عمولات الصيدليات", "REVENUE", "CREDIT"), ("4102", "إيراد رسوم التوصيل", "REVENUE", "CREDIT"), # المصروفات (مدين بطبيعته) ("5101", "مصروف التوصيل والمناديب", "EXPENSE", "DEBIT"), ("5102", "مصروفات تشغيلية وتقنية", "EXPENSE", "DEBIT"), ] self.cursor.executemany( "INSERT OR IGNORE INTO accounts (code, name, acc_type, normal_balance) VALUES (?, ?, ?, ?)", coa ) self.conn.commit() def add_journal_entry(self, reference: str, description: str, lines: list[tuple[str, float, float]], entry_date: str | None = None): """ إضافة قيد يومية مزدوج مع التحقق من توازن المدين والدائن. lines: قائمة من (account_code, debit, credit) """ total_debit = round(sum(x[1] for x in lines), 2) total_credit = round(sum(x[2] for x in lines), 2) if total_debit <= 0 or total_credit <= 0: raise ValueError("يجب أن يحتوي القيد على مبالغ مدين ودائن أكبر من الصفر.") if total_debit != total_credit: raise ValueError(f"القيد غير متوازن! المدين={total_debit} بينما الدائن={total_credit}") # التحقق من وجود الحسابات for code, _, _ in lines: row = self.cursor.execute("SELECT code FROM accounts WHERE code = ?", (code,)).fetchone() if not row: raise ValueError(f"رمز الحساب غير موجود في دليل الحسابات: {code}") date_str = entry_date or datetime.now().strftime("%Y-%m-%d") self.cursor.execute( "INSERT INTO journal_entries (entry_date, reference, description) VALUES (?, ?, ?)", (date_str, reference, description) ) entry_id = self.cursor.lastrowid for code, debit, credit in lines: self.cursor.execute( "INSERT INTO journal_lines (entry_id, account_code, debit, credit) VALUES (?, ?, ?, ?)", (entry_id, code, round(debit, 2), round(credit, 2)) ) self.conn.commit() return entry_id def record_pharmacy_order(self, order_ref: str, medicine_amount: float, delivery_fee: float, commission_rate: float = 0.10, courier_cost: float = 12.0, paid_online: bool = True): """ معالجة طلب دواء متكامل: - العميل يدفع (قيمة الدواء + رسوم التوصيل) - تُخصم عمولة المنصة من نصيب الصيدلية - يُسجل مستحق الصيدلية ومصروف التوصيل """ commission = round(medicine_amount * commission_rate, 2) pharmacy_net = round(medicine_amount - commission, 2) customer_total = round(medicine_amount + delivery_fee, 2) cash_or_receivable = "1101" if paid_online else "1102" lines = [ (cash_or_receivable, customer_total, 0.0), # تحصيل إجمالي الطلب من العميل ("2101", 0.0, pharmacy_net), # صافي مستحق الصيدلية بعد العمولة ("4101", 0.0, commission), # إيراد عمولة المنصة من الصيدلية ("4102", 0.0, delivery_fee), # إيراد رسوم التوصيل ("5101", courier_cost, 0.0), # تكلفة/مصروف التوصيل ("2102", 0.0, courier_cost), # مستحق للمندوب/شركة التوصيل ] return self.add_journal_entry( reference=order_ref, description=f"طلب دواء #{order_ref} (دواء={medicine_amount}, توصيل={delivery_fee}, عمولة={commission})", lines=lines ) def get_account_balances(self) -> dict[str, dict]: rows = self.cursor.execute(""" SELECT a.code, a.name, a.acc_type, a.normal_balance, COALESCE(SUM(jl.debit), 0.0) AS total_debit, COALESCE(SUM(jl.credit), 0.0) AS total_credit FROM accounts a LEFT JOIN journal_lines jl ON a.code = jl.account_code GROUP BY a.code, a.name, a.acc_type, a.normal_balance ORDER BY a.code """).fetchall() balances = {} for code, name, acc_type, normal_bal, dr, cr in rows: # الرصيد الصافي حسب طبيعة الحساب (أو يمكن عرضه كمدين/دائن في ميزان المراجعة) net = (dr - cr) if normal_bal == "DEBIT" else (cr - dr) balances[code] = { "code": code, "name": name, "acc_type": acc_type, "normal_balance": normal_bal, "debit": round(dr, 2), "credit": round(cr, 2), "net": round(net, 2), } return balances def print_trial_balance(self): # 2. ميزان المراجعة (Trial Balance) balances = self.get_account_balances() sum_dr = 0.0 sum_cr = 0.0 print("\n=== ميزان المراجعة (Trial Balance) ===") print(f"{'الرمز':<6} | {'اسم الحساب':<30} | {'مدين':>10} | {'دائن':>10}") print("-" * 66) for b in balances.values(): # نعرض الرصيد النهائي في خانة المدين أو الدائن حسب صافي الحركة diff = round(b["debit"] - b["credit"], 2) dr_val = diff if diff > 0 else 0.0 cr_val = abs(diff) if diff < 0 else 0.0 sum_dr += dr_val sum_cr += cr_val if dr_val > 0 or cr_val > 0: print(f"{b['code']:<6} | {b['name']:<30} | {dr_val:>10.2f} | {cr_val:>10.2f}") print("-" * 66) print(f"{'الإجمالي':<39} | {sum_dr:>10.2f} | {sum_cr:>10.2f}") print(f"حالة الميزان: {'متوازن ✅' if round(sum_dr, 2) == round(sum_cr, 2) else 'غير متوازن ❌'}") def print_financial_statements(self): # 3. القوائم المالية: قائمة الدخل + الميزانية العمومية balances = self.get_account_balances() revenues = [b for b in balances.values() if b["acc_type"] == "REVENUE"] expenses = [b for b in balances.values() if b["acc_type"] == "EXPENSE"] assets = [b for b in balances.values() if b["acc_type"] == "ASSET"] liabilities = [b for b in balances.values() if b["acc_type"] == "LIABILITY"] equities = [b for b in balances.values() if b["acc_type"] == "EQUITY"] total_revenue = round(sum(b["net"] for b in revenues), 2) total_expense = round(sum(b["net"] for b in expenses), 2) net_income = round(total_revenue - total_expense, 2) total_assets = round(sum(b["net"] for b in assets), 2) total_liabilities = round(sum(b["net"] for b in liabilities), 2) total_equity_base = round(sum(b["net"] for b in equities), 2) total_equity_with_ni = round(total_equity_base + net_income, 2) print("\n=== قائمة الدخل (Income Statement) ===") for r in revenues: if r["net"] != 0: print(f" + إيراد ({r['name']}): {r['net']:.2f}") print(f"إجمالي الإيرادات: {total_revenue:.2f}") for e in expenses: if e["net"] != 0: print(f" - مصروف ({e['name']}): {e['net']:.2f}") print(f"إجمالي المصروفات: {total_expense:.2f}") print(f"صافي الربح / (الخسارة): {net_income:.2f}") print("\n=== الميزانية العمومية (Balance Sheet) ===") print("الأصول:") for a in assets: if a["net"] != 0: print(f" - {a['name']}: {a['net']:.2f}") print(f"إجمالي الأصول: {total_assets:.2f}") print("الخصوم وحقوق الملكية:") for l in liabilities: if l["net"] != 0: print(f" - {l['name']}: {l['net']:.2f}") for eq in equities: if eq["net"] != 0: print(f" - {eq['name']}: {eq['net']:.2f}") print(f" - صافي أرباح الفترة المرحلة: {net_income:.2f}") l_plus_eq = round(total_liabilities + total_equity_with_ni, 2) print(f"إجمالي الخصوم وحقوق الملكية: {l_plus_eq:.2f}") print(f"معادلة الميزانية: {'متوازنة ✅' if round(total_assets, 2) == l_plus_eq else 'غير متوازنة ❌'}") if __name__ == "__main__": erp = PharmacyDeliveryERP() # 1. قيد افتتاحي: رأس مال نقدي في الصندوق/البنك erp.add_journal_entry( reference="JV-001", description="تأسيس رأس المال النقدي للمنصة", lines=[ ("1101", 5000.0, 0.0), ("3101", 0.0, 5000.0), ] ) # 2. طلب دواء أول: دواء بـ 200، توصيل بـ 20، عمولة المنصة 10%، تكلفة المندوب 12 erp.record_pharmacy_order( order_ref="ORD-1001", medicine_amount=200.0, delivery_fee=20.0, commission_rate=0.10, courier_cost=12.0, paid_online=True ) # 3. طلب دواء ثانٍ: دواء بـ 100، توصيل بـ 15، عمولة المنصة 10%، تكلفة المندوب 10 erp.record_pharmacy_order( order_ref="ORD-1002", medicine_amount=100.0, delivery_fee=15.0, commission_rate=0.10, courier_cost=10.0, paid_online=True ) # 4. سداد جزء من مستحقات الصيدلية erp.add_journal_entry( reference="PAY-001", description="سداد دفعة للصيدلية من رصيد المنصة", lines=[ ("2101", 150.0, 0.0), ("1101", 0.0, 150.0), ] ) # طباعة ميزان المراجعة والقوائم المالية erp.print_trial_balance() erp.print_financial_statements()
