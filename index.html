class SAPFicoERP: def __init__(self): # الأرصدة الافتتاحية self.accounts = { "البنك": 50000.0, "المخزون": 0.0, "الموردين": 0.0, "العملاء": 0.0, "إيرادات المبيعات": 0.0, "تكلفة البضاعة المباعة": 0.0, "مصاريف الشحن": 0.0 } self.inventory_qty = 0 self.item_cost = 0.0 def process_purchase(self, qty, unit_price, shipping_cost): """دورة المشتريات (P2P): توزيع مصاريف الشحن وحساب التكلفة الفعلية""" total_purchase = qty * unit_price total_cost_with_shipping = total_purchase + shipping_cost # حساب التكلفة الفعلية للقطعة الواحدة (تضمين الشحن) self.item_cost = total_cost_with_shipping / qty self.inventory_qty += qty # القيود المحاسبية self.accounts["المخزون"] += total_cost_with_shipping self.accounts["البنك"] -= total_cost_with_shipping print(f"--- [1] تمت دورة المشتريات بنجاح ---") print(f"تكلفة القطعة الفعلية بعد الشحن: {self.item_cost:.2f} د.ل\n") def process_sale(self, sold_qty, selling_price): """دورة المبيعات (O2C): إخراج البضاعة وحساب الأرباح""" if sold_qty > self.inventory_qty: print("خطأ: الكمية المطلوبة غير متوفرة في المخزن!") return total_sales = sold_qty * selling_price cogs = sold_qty * self.item_cost # تكلفة البضاعة المباعة # خصم من المخزون وتحديث الحسابات self.inventory_qty -= sold_qty self.accounts["المخزون"] -= cogs self.accounts["البنك"] += total_sales self.accounts["إيرادات المبيعات"] += total_sales self.accounts["تكلفة البضاعة المباعة"] += cogs print(f"--- [2] تمت دورة المبيعات بنجاح ---") print(f"إجمالي قيمة الفاتورة: {total_sales} د.ل\n") def show_trial_balance(self): """عرض ميزان المراجعة وقائمة الدخل (SAP Trial Balance)""" print("="*40) print(" ميزان المراجعة اللحظي (ERP)") print("="*40) for acc, balance in self.accounts.items(): print(f"- {acc}: {balance:,.2f} د.ل") print("="*40) net_profit = self.accounts["إيرادات المبيعات"] - self.accounts["تكلفة البضاعة المباعة"] print(f"الرصيد المتبقي بالمخزن: {self.inventory_qty} قطعة") print(f"صافي الربح اللحظي: {net_profit:,.2f} د.ل") print("="*40) # --- تجربة النظام بالقيم التي استخدمناها سابقاً --- erp = SAPFicoERP() # 1. شراء 100 قطعة بسعر 20 د.ل ومصاريف شحن 200 د.ل erp.process_purchase(qty=100, unit_price=20.0, shipping_cost=200.0) # 2. بيع 50 قطعة بسعر 35 د.ل للقطعة erp.process_sale(sold_qty=50, selling_price=35.0) # 3. طباعة التقرير النهائي erp.show_trial_balance()
