/** * نظام المحاسبة المتكامل للشركات (الجرد المستمر والمعايير الدولية IFRS) * بلغة JavaScript (Node.js) مع حفظ دائم للبيانات باستخدام SQLite. */ const Database = require('better-sqlite3'); const db = new Database('accounting_system.db'); // ========================================== // 0. طبقة قاعدة البيانات للحفظ الدائم // ========================================== db.exec(` CREATE TABLE IF NOT EXISTS journal_entries ( entry_id TEXT PRIMARY KEY, company_id TEXT, description TEXT, date TEXT, is_posted INTEGER ); CREATE TABLE IF NOT EXISTS journal_lines ( id INTEGER PRIMARY KEY AUTOINCREMENT, entry_id TEXT, account_code TEXT, account_name TEXT, debit REAL, credit REAL, FOREIGN KEY (entry_id) REFERENCES journal_entries (entry_id) ); `); // ========================================== // 1. دليل الحسابات القياسي // ========================================== const AccountType = { ASSET: "الأصول", LIABILITY: "الالتزامات", EQUITY: "حقوق الملكية", REVENUE: "الإيرادات", EXPENSE: "المصروفات" }; const DEFAULT_CHART_OF_ACCOUNTS = { "1101": { name: "النقدية بالخزينة (نقدي)", type: AccountType.ASSET }, "1102": { name: "البنك / المصرف", type: AccountType.ASSET }, "1103": { name: "المدينون (العملاء)", type: AccountType.ASSET }, "1104": { name: "مخزون البضاعة (جرد مستمر)", type: AccountType.ASSET }, "1105": { name: "بضاعة في الطريق (استيراد/شحن)", type: AccountType.ASSET }, "1106": { name: "ذمم شركات الشحن والمندوبين", type: AccountType.ASSET }, "2101": { name: "الدائنون (الموردون)", type: AccountType.LIABILITY }, "2102": { name: "تأمينات وعرابين مقبوضة", type: AccountType.LIABILITY }, "2103": { name: "حساب المقاصة الشريك/التسويات", type: AccountType.LIABILITY }, "3101": { name: "رأس المال", type: AccountType.EQUITY }, "3102": { name: "جاري المالك / الشركاء", type: AccountType.EQUITY }, "4101": { name: "إيرادات المبيعات", type: AccountType.REVENUE }, "5101": { name: "تكلفة البضاعة المباعة (COGS)", type: AccountType.EXPENSE }, "5102": { name: "مسموحات وتسويات المبيعات", type: AccountType.EXPENSE }, "5103": { name: "مصروفات التشغيل والشحن", type: AccountType.EXPENSE } }; // ========================================== // 2. المحرك المحاسبي وتوازن القيود // ========================================== class JournalEntry { constructor(entryId, companyId, description, date = new Date()) { this.entryId = entryId; this.companyId = companyId; this.description = description; this.date = date; this.lines = []; this.isPosted = false; } addLine(accountCode, accountName, debit = 0.0, credit = 0.0) { this.lines.push({ accountCode, accountName, debit: Math.round(debit * 1000) / 1000, credit: Math.round(credit * 1000) / 1000 }); } validateAndPost() { let totalDebit = this.lines.reduce((sum, line) => sum + line.debit, 0); let totalCredit = this.lines.reduce((sum, line) => sum + line.credit, 0); if (Math.abs(totalDebit - totalCredit) > 0.001) { throw new Error(`فشل ترحيل القيد ${this.entryId}: القيد غير متزن! إجمالي المدين (${totalDebit}) لا يساوي الدائن (${totalCredit}).`); } this.isPosted = true; this.saveToDatabase(); return true; } saveToDatabase() { const insertEntry = db.prepare(` INSERT OR REPLACE INTO journal_entries (entry_id, company_id, description, date, is_posted) VALUES (?, ?, ?, ?, ?) `); const deleteLines = db.prepare(`DELETE FROM journal_lines WHERE entry_id = ?`); const insertLine = db.prepare(` INSERT INTO journal_lines (entry_id, account_code, account_name, debit, credit) VALUES (?, ?, ?, ?, ?) `); const transaction = db.transaction((entry) => { insertEntry.run(entry.entryId, entry.companyId, entry.description, entry.date.toISOString(), entry.isPosted ? 1 : 0); deleteLines.run(entry.entryId); for (let line of entry.lines) { insertLine.run(entry.entryId, line.accountCode, line.accountName, line.debit, line.credit); } }); transaction(this); } } // ========================================== // 3. إدارة أوامر الشراء والمبيعات // ========================================== class PurchaseManager { static createPurchaseOrder(entryId, companyId, purchaseType, paymentMethod, amount) { let entry = new JournalEntry(entryId, companyId, `أمر شراء (${purchaseType}) - الدفع: ${paymentMethod}`); let targetCode = purchaseType === "محلي" ? "1104" : "1105"; let targetName = purchaseType === "محلي" ? "مخزون البضاعة (جرد مستمر)" : "بضاعة في الطريق (استيراد/شحن)"; let paymentMap = { "نقدي": ["1101", "النقدية بالخزينة (نقدي)"], "مصرف": ["1102", "البنك / المصرف"], "دائنون": ["2101", "الدائنون (الموردون)"], "جاري المالك": ["3102", "جاري المالك / الشركاء"] }; let [payCode, payName] = paymentMap[paymentMethod]; entry.addLine(targetCode, targetName, amount, 0.0); entry.addLine(payCode, payName, 0.0, amount); entry.validateAndPost(); return entry; } static settleImportedGoods(entryId, companyId, amount) { let entry = new JournalEntry(entryId, companyId, "تسوية بضاعة في الطريق إلى المخزون الرئيسي"); entry.addLine("1104", "مخزون البضاعة (جرد مستمر)", amount, 0.0); entry.addLine("1105", "بضاعة في الطريق (استيراد/شحن)", 0.0, amount); entry.validateAndPost(); return entry; } } class SalesManager { static createSalesOrder(entryId, companyId, paymentMethod, receiptType, orderStatus, collector, saleAmount, costAmount, advancePaid = 0.0) { let entries = []; let salesEntry = new JournalEntry(entryId, companyId, `أمر بيع (${orderStatus}) - تحصيل: ${collector}`); let debitCode, debitName; if (["شركات الشحن", "مندوبي الشركة"].includes(collector)) { debitCode = "1106"; debitName = `ذمم ${collector}`; } else { let paymentMap = { "نقدي": ["1101", "النقدية بالخزينة (نقدي)"], "مصرف": ["1102", "البنك / المصرف"], "مدينون": ["1103", "المدينون (العملاء)"], "مقاصة": ["2103", "حساب المقاصة الشريك/التسويات"] }; [debitCode, debitName] = paymentMap[paymentMethod]; } if (receiptType === "كامل") { salesEntry.addLine(debitCode, debitName, saleAmount, 0.0); salesEntry.addLine("4101", "إيرادات المبيعات", 0.0, saleAmount); } else if (["جزئي", "عربون"].includes(receiptType)) { let remaining = saleAmount - advancePaid; if (advancePaid > 0) salesEntry.addLine(debitCode, debitName, advancePaid, 0.0); if (remaining > 0) salesEntry.addLine("1103", "المدينون (العملاء)", remaining, 0.0); salesEntry.addLine("4101", "إيرادات المبيعات", 0.0, saleAmount); } salesEntry.validateAndPost(); entries.push(salesEntry); // قيد تكلفة البضاعة المباعة (جرد مستمر) let cogsEntry = new JournalEntry(`${entryId}-COGS`, companyId, "إثبات تكلفة البضاعة المباعة آلياً (جرد مستمر)"); cogsEntry.addLine("5101", "تكلفة البضاعة المباعة (COGS)", costAmount, 0.0); cogsEntry.addLine("1104", "مخزون البضاعة (جرد مستمر)", 0.0, costAmount); cogsEntry.validateAndPost(); entries.push(cogsEntry); return entries; } static createSalesAdjustment(entryId, companyId, adjustmentAmount, reason) { let entry = new JournalEntry(entryId, companyId, `تسويات مبيعات: ${reason}`); entry.addLine("5102", "مسموحات وتسويات المبيعات", adjustmentAmount, 0.0); entry.addLine("1103", "المدينون (العملاء)", 0.0, adjustmentAmount); entry.validateAndPost(); return entry; } } // ========================================== // 4. النظام المحاسبي وإصدار التقارير // ========================================== class AccountingSystem { constructor(companyId) { this.companyId = companyId; } getJournalLedger() { const stmt = db.prepare(`SELECT entry_id, description, date, is_posted FROM journal_entries WHERE company_id = ?`); const entriesRows = stmt.all(this.companyId); let entries = []; for (let row of entriesRows) { let entry = new JournalEntry(row.entry_id, this.companyId, row.description, new Date(row.date)); entry.isPosted = Boolean(row.is_posted); const lineStmt = db.prepare(`SELECT account_code, account_name, debit, credit FROM journal_lines WHERE entry_id = ?`); let linesRows = lineStmt.all(row.entry_id); for (let l of linesRows) { entry.addLine(l.account_code, l.account_name, l.debit, l.credit); } entries.push(entry); } return entries; } postOpeningBalance(balances) { let entry = new JournalEntry("OPENING-001", this.companyId, "القيد الافتتاحي للشركة"); for (let [accCode, val] of Object.entries(balances)) { let accName = DEFAULT_CHART_OF_ACCOUNTS[accCode] ? DEFAULT_CHART_OF_ACCOUNTS[accCode].name : "حساب غير معرف"; entry.addLine(accCode, accName, val.debit || 0.0, val.credit || 0.0); } entry.validateAndPost(); return entry; } generateAmericanJournal() { console.log(`\n==================== دفتر اليومية الأمريكية - الشركة: ${this.companyId} ====================`); let entries = this.getJournalLedger(); for (let entry of entries) { for (let line of entry.lines) { console.log(`[${entry.entryId}] | ${entry.description} | الحساب: ${line.accountName} | مدين: ${line.debit} | دائن: ${line.credit}`); } } console.log("=========================================================================="); } } // ========================================== // 5. تجربة وتشغيل المنظومة // ========================================== try { const company = new AccountingSystem("CORP-JS-2026"); // إنشاء القيد الافتتاحي company.postOpeningBalance({ "1101": { debit: 150000.0, credit: 0.0 }, "3101": { debit: 0.0, credit: 150000.0 } }); // أوامر الشراء والبيع والتسويات PurchaseManager.createPurchaseOrder("PO-JS-1", "CORP-JS-2026", "استيراد", "مصرف", 40000.0); PurchaseManager.settleImportedGoods("SETTLE-JS-1", "CORP-JS-2026", 40000.0); SalesManager.createSalesOrder("SO-JS-1", "CORP-JS-2026", "نقدي", "عربون", "فوري", "شركات الشحن", 25000.0, 15000.0, 5000.0); SalesManager.createSalesAdjustment("ADJ-JS-1", "CORP-JS-2026", 300.0, "مسموحات مبيعات"); console.log("[+] تمت عمليات الحفظ الدائم واستخراج اليومية بنجاح عبر JavaScript!"); company.generateAmericanJournal(); } catch (error) { console.error("خطأ في النظام المحاسبي:", error.message); }
