from __future__ import annotations from dataclasses import dataclass, field from datetime import date from decimal import Decimal, ROUND_HALF_UP from enum import Enum from typing import Dict, List def money(value: float | int | str | Decimal) -> Decimal: return Decimal(str(value)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP) class AccountType(str, Enum): ASSET = "أصول" LIABILITY = "خصوم" REVENUE = "إيرادات" EXPENSE = "مصروفات" @dataclass class Account: code: str name: str account_type: AccountType balance: Decimal = Decimal("0.00") def apply(self, debit: Decimal, credit: Decimal) -> None: #
