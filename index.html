from __future__ import annotations from dataclasses import dataclass, field from datetime import date from enum import Enum from typing import Dict, List, Optional class AccountType(str, Enum): ASSET = "أصول" # Assets LIABILITY = "خصوم" # Liabilities EQUITY = "حقوق ملكية" # Equity REVENUE = "إيرادات" # Revenue EXPENSE = "مصروفات" # Expenses class NormalSide(str, Enum): DEBIT = "مدين" CREDIT = "دائن" @dataclass class Account: code: str name: str account_type: AccountType normal_side: NormalSide parent_code: Optional[str] = None @dataclass class JournalLine: account_code: str debit: float = 0.0 credit: float = 0.0 memo: str = "" @dataclass class JournalEntry: entry_id: str entry_date: date description: str reference: str lines: List[JournalLine] = field(default_factory=list) def validate(self) -> None: if not self.lines: raise ValueError(f"القيد {self.entry_id} لا يحتوي على أي سطور.") total_debit = round(sum(line.debit for line in self.lines), 2) total_credit = round(sum(line.credit for line in self.lines), 2) if total_debit <= 0 or total_credit <= 0: raise ValueError(f"قيم القيد {self.entry_id} يجب أن تكون أكبر من الصفر.") if total_debit != total_credit: raise ValueError( f"القيد {self.entry_id} غير متوازن! مدين={total_debit} بينما دائن={total_credit}" ) class PharmacyDeliveryERP: def __init__(self) -> None: self.accounts: Dict[str, Account] = {} self.journal_entries: List[JournalEntry] = [] self._init_chart_of_accounts() def _init_chart_of_accounts(self) -> None: """دليل الحسابات (Chart of Accounts) مستوحى من معايير ERP.""" coa = [ # 1xxx: الأصول (Assets) - ------------------------------------------------------------------ def trial_balance(self) -> List[dict]: """ميزان المراجعة (Trial Balance).""" rows = [] for code, acc in bal != 0: expenses[acc.name] = bal total_rev = round(sum(revenues.values()), 2) total_exp = round(sum(expenses.values()), 2) net_income = round(total_rev - total_exp, 2) return { "revenues": revenues, "total_revenue": total_rev, "expenses": expenses, "total_expense": total_exp, "net_income": net_income, } def balance_sheet(self) -> dict: """الميزانية العمومية / قائمة المركز المالي (Balance Sheet).""" assets = {} liabilities = {} equity = {} for code, acc in sorted(self.accounts.items()): bal = self.get_account_balance(code) if acc.account_type == AccountType.ASSET and bal != 0: assets[acc.name] = bal elif acc.account_type == AccountType.LIABILITY and bal != 0: liabilities[acc.name] = bal elif acc.account_type == AccountType.EQUITY and bal != 0: equity[acc.name] = bal # ترحيل صافي الربح/الخسارة إلى حقوق الملكية لإغلاق الميزانية net_income = self.income_statement()["net_income"] equity["صافي ربح (خسارة) الفترة الحالية"] = net_income total_assets = round(sum(assets.values()), 2) total_liabilities = round(sum(liabilities.values()), 2) total_equity = round(sum(equity.values()), 2) return { "assets": assets, "total_assets": total_assets, "liabilities": liabilities, "total_liabilities": total_liabilities, "equity": equity, "total_equity": total_equity, "liabilities_plus_equity": round(total_liabilities + total_equity, 2), } #
