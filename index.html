إنشاء كود المنتج تلقائياً cursor.execute("SELECT COUNT(*) FROM products") count = cursor.fetchone()[0] + 1 code = f"PRD-{count:04d}" cursor.execute( "INSERT INTO products (code, name, weight, color, price) VALUES" " (?, ?, ?, ?, ?)", (code, e_name.get(), weight, e_color.get(), price), ) conn.commit() messagebox.SUCCESS( "نجاح", f"تم إضافة المنتج بنجاح، والكود المنشأ هو: {code}" ) refresh_table() tk.Button( form_frame, text="حفظ المنتج", bg="green", fg="white", command=save_product, ).grid(row=1, column=0, columnspan=2, pady=5) # جدول عرض المنتجات cols = ("الكود", "الوصف", "الوزن", "اللون", "السعر", "التكلفة", "المخزون") tree = ttk.Treeview(frame, columns=cols, show="headings") for col in cols: tree.heading(col, text=col) tree.pack(fill="both", expand=True, pady=10) def refresh_table(): for row in tree.get_children(): tree.delete(row) cursor.execute( "SELECT code, name, weight, color, price, cost, stock FROM products" ) for row in cursor.fetchall(): tree.insert("", "end", values=row) refresh_table() # --- المشتريات والشحن المتقدم --- def open_purchases(self): self.clear_window() self.show_main_dashboard() frame = tk.Frame(self.root) frame.pack(fill="both", expand=True, padx=20, pady=20) tk.Label( frame, text="إدارة المشتريات والشحن والجمرك", font=("Arial", 14, "bold") ).pack(anchor="e", pady=5) # نموذج طلب الشراء f_frame = tk.Frame(frame) f_frame.pack(fill="x", pady=5) tk.Label(f_frame, text="رقم طلب الشراء:").grid(row=0, column=5) e_po = tk.Entry(f_frame) e_po.grid(row=0, column=4) tk.Label(f_frame, text="المورد:").grid(row=0, column=3) e_sup = tk.Entry(f_frame) e_sup.grid(row=0, column=2) tk.Label(f_frame, text="طريقة السداد:").grid(row=0, column=1) pay_method = ttk.Combobox( f_frame, values=["نقداً", "مصرف", "آجل"], state="readonly" ) pay_method.grid(row=0, column=0) pay_method.current(0) # إضافة أصناف لطلب الشراء items_frame = tk.LabelFrame( frame, text="أصناف الشحنة في الطريق / المستلمة" ) items_frame.pack(fill="x", pady=10) tk.Label(items_frame, text="كود المنتج:").grid(row=0, column=0) e_pcode = tk.Entry(items_frame) e_pcode.grid(row=0, column=1) tk.Label(items_frame, text="الكمية:").grid(row=0, column=2) e_qty = tk.Entry(items_frame) e_qty.grid(row=0, column=3) tk.Label(items_frame, text="سعر الشراء الفردي:").grid(row=0, column=4) e_pprice = tk.Entry(items_frame) e_pprice.grid(row=0, column=5) temp_items = [] def add_item_to_list(): code = e_pcode.get() try: qty = float(e_qty.get()) price = float(e_pprice.get()) except ValueError: messagebox.showerror("خطأ", "الكمية والسعر يجب أن يكونا أرقاماً") return cursor.execute("SELECT weight FROM products WHERE code=?", (code,)) res = cursor.fetchone() if not res: messagebox.showerror("خطأ", "كود المنتج غير موجود!") return weight = res[0] * qty temp_items.append((code, qty, price, weight)) messagebox.showinfo("تم", "تمت الإضافة للقائمة المؤقتة للشراء") tk.Button( items_frame, text="إضافة للصنف", command=add_item_to_list ).grid(row=0, column=6, padx=10) def submit_po(): if not temp_items: messagebox.showerror("خطأ", "لم تتم إضافة أي أصناف!") return total_goods = sum([i[1] * i[2] for i in temp_items]) po_no = e_po.get() # تسجيل قيد المشتريات الآلي (من ح/ البضاعة في الطريق أو المخزون إلى ح/ المورد أو الصندوق) cursor.execute( "INSERT INTO purchase_orders (po_no, supplier, status, total_goods)" " VALUES (?, ?, 'InTransit', ?)", (po_no, e_sup.get(), total_goods), ) po_id = cursor.lastrowid for item in temp_items: cursor.execute( "INSERT INTO purchase_items (po_id, product_code, qty, price," " weight) VALUES (?, ?, ?, ?, ?)", (po_id, item[0], item[1], item[2], item[3]), ) # القيد المحاسبي التلقائي p_type = pay_method.get() cr_acc = ( "211" if p_type == "آجل" else ("111" if p_type == "نقداً" else "112") ) date_now = datetime.now().strftime("%Y-%m-%d") cursor.execute( "INSERT INTO journal (date, entry_no, account_code, description," " debit, credit) VALUES (?, ?, '122', ?, ?, 0)", (date_now, po_no, f"اثبات بضاعة في الطريق لطلب {po_no}", total_goods), ) cursor.execute( "INSERT INTO journal (date, entry_no, account_code, description," " debit, credit) VALUES (?, ?, ?, ?, 0, ?)", (date_now, po_no, cr_acc, f"سداد/التزام طلب شراء {po_no}", total_goods), ) conn.commit() messagebox.showinfo( "نجاح", "تم إصدار أمر الشراء وتسجيل البضاعة في الطريق وتوليد القيود أوتوماتيكياً!", ) tk.Button( frame, text="اعتماد طلب الشراء", bg="blue", fg="white", command=submit_po, ).pack(pady=5) # معالجة وصول الشحنة ومصاريف الشحن والجمرك الموزعة أوتوماتيكياً dist_frame = tk.LabelFrame( frame, text="استلام الشحنة وتوزيع مصاريف الشحن الدولي والجمارك أوتوماتيكياً" ) dist_frame.pack(fill="x", pady=10) tk.Label(dist_frame, text="رقم طلب الشراء المراد استلامه:").grid( row=0, column=0 ) e_po_rcv = tk.Entry(dist_frame) e_po_rcv.grid(row=0, column=1) tk.Label( dist_frame, text="إجمالي مصاريف الشحن الدولي (توزع أوتوماتيك بالوزن):" ).grid(row=1, column=0) e_ship = tk.Entry(dist_frame) e_ship.grid(row=1, column=1) tk.Label(dist_frame, text="إجمالي مصاريف الجمرك (توزع أوتوماتيك بالنسبة):") \ .grid(row=2, column=0) e_customs = tk.Entry(dist_frame) e_customs.grid(row=2, column=1) def process_arrival(): po_no = e_po_rcv.get() try: ship_cost = float(e_ship.get() or 0) customs_cost = float(e_customs.get() or 0) except ValueError: messagebox.showerror("خطأ", "المصاريف يجب أن تكون أرقاماً") return cursor.execute("SELECT id, total_goods FROM purchase_orders WHERE po_no=?", (po_no,)) po_res = cursor.fetchone() if not po_res: messagebox.showerror("خطأ", "رقم الطلب غير موجود!") return po_id, total_goods = po_res # جلب أصناف الطلب وأوزانها cursor.execute("SELECT product_code, qty, price, weight FROM purchase_items WHERE po_id=?", (po_id,)) items = cursor.fetchall() total_weight = sum([i[3] for i in items]) date_now = datetime.now().strftime("%Y-%m-%d") # توزيع الشحن الدولي والجمارك على تكلفة المنتجات وتحديث المخزون for item in items: p_code, qty, base_price, item_weight = item # حصة الصنف من الشحن بناء على الوزن item_ship = (item_weight / total_weight) * ship_cost if total_weight > 0 else 0 # حصة الصنف من الجمارك بناء على القيمة النسبية item_customs = ( (qty * base_price) / total_goods ) * customs_cost if total_goods > 0 else 0 # التكلفة الجديدة للقطعة الواحدة = السعر الأساسي + (نصيبها من الشحن والجمارك ÷ الكمية) new_unit_cost = base_price + ((item_ship + item_customs) / qty) # تحديث تكلفة المنتج في المخزون وزيادة الكمية cursor.execute("UPDATE products SET cost=?, stock = stock + ? WHERE code=?", (new_unit_cost, qty, p_code)) # قيد إثبات مصاريف الشحن والجمارك وسدادها cursor.execute("INSERT INTO journal (date, entry_no, account_code, description, debit, credit) VALUES (?, ?, '121', ?, ?, 0)", (date_now, po_no, f"اقفال البضاعة في الطريق والمصاريف للمخزون - {po_no}", ship_cost + customs_cost)) cursor.execute("INSERT INTO journal (date, entry_no, account_code, description, debit, credit) VALUES (?, ?, '112', ?, 0, ?)", (date_now, po_no, f"سداد مصاريف شحن وجمرك - {po_no}", ship_cost + customs_cost)) cursor.execute("UPDATE purchase_orders SET status='Received', shipping_cost=?, customs_cost=? WHERE id=?", (ship_cost, customs_cost, po_id)) conn.commit() messagebox.showinfo("نجاح", "تم استلام البضاعة وتوزيع الشحن والجمارك أوتوماتيكياً على تكلفة المخزون وتوليد القيود المحاسبية!") tk.Button(dist_frame, text="تأكيد الاستلام وتوزيع التكاليف", bg="orange", fg="black", command=process_arrival).grid(row=3, column=0, columnspan=2, pady=10) # --- المخزون --- def open_inventory(self): self.open_products() # يعرض شاشة المنتجات والمخزون معاً # --- المبيعات --- def open_sales(self): self.clear_window() self.show_main_dashboard() frame = tk.Frame(self.root) frame.pack(fill="both", expand=True, padx=20, pady=20) tk.Label(frame, text="إدارة المبيعات وإصدار الفواتير", font=("Arial", 14, "bold")).pack(anchor="e", pady=5) f = tk.Frame(frame) f.pack(fill="x", pady=10) tk.Label(f, text="رقم الفواتير:").grid(row=0, column=0) e_inv = tk.Entry(f) e_inv.grid(row=0, column=1) tk.Label(f, text="كود المنتج:").grid(row=0, column=2) e_pcode = tk.Entry(f) e_pcode.grid(row=0, column=3) tk.Label(f, text="الكمية المباعة:").grid(row=0, column=4) e_qty = tk.Entry(f) e_qty.grid(row=0, column=5) def submit_sale(): inv_no = e_inv.get() p_code = e_pcode.get() try: qty = float(e_qty.get()) except ValueError: messagebox.showerror("خطأ", "الكمية غير صحيحة") return cursor.execute("SELECT price, cost, stock FROM products WHERE code=?", (p_code,)) res = cursor.fetchone() if not res: messagebox.showerror("خطأ", "المنتج غير موجود") return price, cost, stock = res if stock < qty: messagebox.showerror("خطأ", f"الكمية غير متوفرة في المخزون! المتاح: {stock}") return total_sale = price * qty total_cost = cost * qty date_now = datetime.now().strftime("%Y-%m-%d") # خصم المخزون cursor.execute("UPDATE products SET stock = stock - ? WHERE code=?", (qty, p_code)) # القيود المحاسبية التلقائية للمبيعات # 1. إثبات الإيراد cursor.execute("INSERT INTO journal (date, entry_no, account_code, description, debit, credit) VALUES (?, ?, '111', ?, ?, 0)", (date_now, inv_no, f"قيمة مبيعات فاتورة {inv_no}", total_sale)) cursor.execute("INSERT INTO journal (date, entry_no, account_code, description, debit, credit) VALUES (?, ?, '411', ?, 0, ?)", (date_now, inv_no, f"إيراد مبيعات {inv_no}", total_sale)) # 2. إثبات تكلفة البضاعة المباعة وخفض المخزون cursor.execute("INSERT INTO journal (date, entry_no, account_code, description, debit, credit) VALUES (?, ?, '511', ?, ?, 0)", (date_now, inv_no, f"تكلفة بضاعة مباعة {inv_no}", total_cost)) cursor.execute("INSERT INTO journal (date, entry_no, account_code, description, debit, credit) VALUES (?, ?, '121', ?, 0, ?)", (date_now, inv_no, f"خروج مخزون فاتورة {inv_no}", total_cost)) conn.commit() messagebox.showinfo("نجاح", "تمت عملية البيع وتوليد قيود المبيعات وتكلفة البضاعة المباعة تلقائياً!") tk.Button(f, text="إتمام البيع وتوليد القيود", bg="blue", fg="white", command=submit_sale).grid(row=0, column=6, padx=10) # --- المحاسبة والتقارير --- def open_accounting(self): self.clear_window() self.show_main_dashboard() frame = tk.Frame(self.root) frame.pack(fill="both", expand=True, padx=20, pady=20) # تبويبات فرعية للمحاسبة notebook = ttk.Notebook(frame) notebook.pack(fill="both", expand=True) # 2. دفتر اليومية journal_frame = ttk.Frame(notebook) notebook.add(journal_frame, text="دفتر اليومية") j_cols = ("التاريخ", "رقم القند/الحركة", "الحساب", "البيان", "مدين", "دائن") j_tree = ttk.Treeview(journal_frame, columns=j_cols, show="headings") for col in j_cols: j_tree.heading(col, text=col) j_tree.pack(fill="both", expand=True) cursor.execute("SELECT j.date, j.entry_no, a.name, j.description, j.debit, j.credit FROM journal j JOIN accounts a ON j.account_code = a.code") for row in cursor.fetchall(): j_tree.insert("", "end", values=row) # 3. ميزان المراجعة tb_frame = ttk.Frame(notebook) notebook.add(tb_frame, text="ميزان المراجعة") tb_cols = ("رقم الحساب", "اسم الحساب", "إجمالي المدين", "إجمالي الدائن", "الرصيد") tb_tree = ttk.Treeview(tb_frame, columns=tb_cols, show="headings") for col in tb_cols: tb_tree.heading(col, text=col) tb_tree.pack(fill="both", expand=True) cursor.execute("SELECT a.code, a.name, SUM(j.debit), SUM(j.credit) FROM accounts a LEFT JOIN journal j ON a.code = j.account_code GROUP BY a.code") for row in cursor.fetchall(): code, name, debit, credit = row debit = debit or 0 credit = credit or 0 balance = debit - credit tb_tree.insert("", "end", values=(code, name, debit, credit, balance)) #
